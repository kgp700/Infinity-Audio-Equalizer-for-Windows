cmake_minimum_required(VERSION 3.21)
project(InfinityCompressor VERSION 0.7.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vst3sdk/CMakeLists.txt")
  message(FATAL_ERROR "Missing vst3sdk; see vst3/README_BUILD.txt for the official recursive SDK checkout")
endif()
add_subdirectory(vst3sdk)
smtg_add_vst3plugin(InfinityCompressor src/plugin.cpp src/compressor_core.h)
target_link_libraries(InfinityCompressor PRIVATE sdk)
target_include_directories(InfinityCompressor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
